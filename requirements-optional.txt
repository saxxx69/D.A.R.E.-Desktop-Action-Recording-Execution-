"""Tests for the DSL generator and Intent Layer."""

from __future__ import annotations

import json
from pathlib import Path

import pytest

from dare.dsl.generator import DSLGenerator, _stdlib_yaml
from dare.intent.classifier import classify_action, IntentClassifier


# ---------------------------------------------------------------------------
# DSL generator
# ---------------------------------------------------------------------------


def _make_run_with_graph(tmp_path: Path, actions: list[dict]) -> Path:
    run = tmp_path / "run"
    (run / "processed").mkdir(parents=True)
    (run / "logs").mkdir(parents=True)
    (run / "processed" / "action_graph.json").write_text(
        json.dumps({"actions": actions}, indent=2),
        encoding="utf-8",
    )
    return run


def _click(aid: str, hint: str, text: str = "BUY") -> dict:
    return {
        "id": aid,
        "type": "click",
        "raw_xy": [100, 100],
        "frame": "s_000000.png",
        "target": {
            "hint": hint, "text": text, "type": "button",
            "element_id": "abc123", "vision": "templates/x.png",
            "relative_position": [0.5, 0.5], "confidence": 0.92,
        },
        "expected_diff": "new_element",
    }


def _type(aid: str, value: str) -> dict:
    return {
        "id": aid,
        "type": "type",
        "value": value,
        "frame": "s_000000.png",
        "target": {
            "hint": "lot_input", "text": "", "type": "input",
            "element_id": "def456",
            "relative_position": [0.3, 0.4], "confidence": 0.85,
        },
        "expected_diff": "text_change",
    }


def test_dsl_emits_wait_click_for_click_actions(tmp_path: Path) -> None:
    run = _make_run_with_graph(tmp_path, [_click("a0", "buy_button")])
    DSLGenerator(run).run()
    text = (run / "processed" / "action.dsl.yaml").read_text("utf-8")
    assert "WAIT_FOR" in text
    assert "CLICK" in text
    assert "buy_button" in text


def test_dsl_emits_type_and_verify_for_type_actions(tmp_path: Path) -> None:
    run = _make_run_with_graph(tmp_path, [_type("a1", "0.25")])
    DSLGenerator(run).run()
    text = (run / "processed" / "action.dsl.yaml").read_text("utf-8")
    assert "TYPE" in text
    assert "0.25" in text
    assert "VERIFY" in text
    assert "text_present:0.25" in text


def test_dsl_full_sequence(tmp_path: Path) -> None:
    run = _make_run_with_graph(
        tmp_path, [_click("a0", "buy_button"), _type("a1", "0.25")]
    )
    summary = DSLGenerator(run).run()
    assert summary["actions"] == 2
    text = (run / "processed" / "action.dsl.yaml").read_text("utf-8")
    # Order: WAIT_FOR, CLICK, TYPE, VERIFY
    assert text.index("WAIT_FOR") < text.index("CLICK")
    assert text.index("CLICK") < text.index("TYPE")
    assert text.index("TYPE") < text.index("VERIFY")


def test_dsl_yaml_round_trips_with_pyyaml(tmp_path: Path) -> None:
    pytest.importorskip("yaml")
    import yaml
    run = _make_run_with_graph(tmp_path, [_click("a0", "x")])
    DSLGenerator(run).run()
    text = (run / "processed" / "action.dsl.yaml").read_text("utf-8")
    loaded = yaml.safe_load(text)
    assert loaded["version"] == "1.0"
    assert "actions" in loaded


def test_stdlib_yaml_basics() -> None:
    """Hand-rolled YAML emitter handles the schema we need."""
    pytest.importorskip("yaml")
    import yaml
    doc = {
        "version": "1.0",
        "actions": [
            {"action": "CLICK", "target": {"hint": "buy_button"}},
            {"action": "TYPE", "value": "0.25"},
        ],
    }
    out = _stdlib_yaml(doc)
    parsed = yaml.safe_load(out)
    assert parsed == doc


def test_dsl_missing_action_graph_raises(tmp_path: Path) -> None:
    run = tmp_path / "run"
    (run / "processed").mkdir(parents=True)
    with pytest.raises(FileNotFoundError):
        DSLGenerator(run).run()

def test_classify_click_with_known_label_is_static() -> None:
    c = classify_action(_click("a0", "buy_button", text="BUY"))
    assert c.type == "STATIC"


def test_classify_click_with_id_only_is_static() -> None:
    a = _click("a0", "x", text="")
    c = classify_action(a)
    assert c.type == "STATIC"


def test_classify_click_with_unknown_target_is_dynamic() -> None:
    a = _click("a0", "x", text="")
    a["target"]["element_id"] = ""
    c = classify_action(a)
    assert c.type == "DYNAMIC"


def test_classify_type_numeric_is_dynamic_with_range() -> None:
    c = classify_action(_type("a0", "0.25"))
    assert c.type == "DYNAMIC"
    assert c.parameters
    p = c.parameters[0]
    assert p.type == "float"
    assert p.range is not None and p.range[0] < p.range[1]
    assert 0.25 in p.example_values or any(abs(v - 0.25) < 1e-6 for v in p.example_values)


def test_classify_type_int_is_dynamic_int() -> None:
    c = classify_action(_type("a0", "10"))
    assert c.type == "DYNAMIC"
    assert c.parameters[0].type == "int"


def test_classify_type_url_is_dynamic_url() -> None:
    c = classify_action(_type("a0", "https://example.com/foo"))
    assert c.type == "DYNAMIC"
    assert c.parameters[0].type == "url"


def test_classify_type_path_is_dynamic_path() -> None:
    c = classify_action(_type("a0", "/var/log/foo.log"))
    assert c.type == "DYNAMIC"
    assert c.parameters[0].type == "path"


def test_classify_type_alphanumeric_is_dynamic_string() -> None:
    c = classify_action(_type("a0", "AAPL5"))
    assert c.type == "DYNAMIC"
    assert c.parameters[0].type == "string"


def test_classify_type_pure_text_is_static() -> None:
    c = classify_action(_type("a0", "hello world"))
    assert c.type == "STATIC"


def test_classifier_batch() -> None:
    actions = [_click("a0", "x"), _type("a1", "0.25"), _type("a2", "fixed text")]
    results = IntentClassifier().classify_all(actions)
    assert [r.type for r in results] == ["STATIC", "DYNAMIC", "STATIC"]

def test_intent_builder_writes_registry_and_markdowns(tmp_path: Path) -> None:
    from dare.intent.builder import IntentBuilder

    run = _make_run_with_graph(
        tmp_path, [_click("a0", "buy_button"), _type("a1", "0.25")]
    )
    summary = IntentBuilder(run).run()
    assert summary["static"] == 1
    assert summary["dynamic"] == 1
    reg = json.loads((run / "processed" / "intent_registry.json").read_text("utf-8"))
    assert "a0" in reg and "a1" in reg
    assert reg["a0"]["type"] == "STATIC"
    assert reg["a1"]["type"] == "DYNAMIC"
    assert "example_values" in reg["a1"]
    # Markdown files exist
    assert (run / "docs" / "intents" / "a0.md").is_file()
    assert (run / "docs" / "intents" / "a1.md").is_file()
    # Action graph annotated with intent_file
    g = json.loads((run / "processed" / "action_graph.json").read_text("utf-8"))
    for a in g["actions"]:
        assert a.get("intent_file", "").startswith("docs/intents/")
