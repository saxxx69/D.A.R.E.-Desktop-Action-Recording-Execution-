"""Tests for the normalizer module."""

from __future__ import annotations

import json
from pathlib import Path

import pytest

from dare.normalizer.element_matcher import (
    bbox_iou,
    bbox_relative,
    fuzzy_text_similarity,
    stable_element_id,
)
from dare.normalizer.state_diff_engine import StateDiffEngine
from dare.normalizer.ui_state_extractor import UIElement


# ---------------------------------------------------------------------------
# matcher
# ---------------------------------------------------------------------------


def test_iou_identical_boxes_is_one() -> None:
    assert bbox_iou((10, 10, 50, 30), (10, 10, 50, 30)) == 1.0


def test_iou_disjoint_boxes_is_zero() -> None:
    assert bbox_iou((0, 0, 10, 10), (100, 100, 10, 10)) == 0.0


def test_iou_partial_overlap() -> None:
    iou = bbox_iou((0, 0, 20, 20), (10, 10, 20, 20))
    # Intersection: 10x10=100; union: 400+400-100=700; IoU ≈ 0.143
    assert 0.10 < iou < 0.20


def test_bbox_relative_stable_under_one_pixel_shift() -> None:
    a = bbox_relative((100, 200, 50, 30), (1000, 1000))
    b = bbox_relative((101, 200, 50, 30), (1000, 1000))
    assert a == b


def test_bbox_relative_handles_zero_size() -> None:
    assert bbox_relative((0, 0, 10, 10), (0, 0)) == (0, 0, 0, 0)


def test_fuzzy_text_handles_empty() -> None:
    assert fuzzy_text_similarity("", "") == 1.0
    assert fuzzy_text_similarity("a", "") == 0.0


def test_fuzzy_text_high_for_close_strings() -> None:
    sim = fuzzy_text_similarity("Buy Now", "BuyNow")
    assert sim > 0.7


def test_stable_id_survives_one_pixel_shift() -> None:
    s = (1000, 1000)
    a = stable_element_id("BUY", (100, 100, 80, 30), "button", s)
    b = stable_element_id("BUY", (101, 100, 80, 30), "button", s)
    assert a == b


def test_stable_id_changes_with_text() -> None:
    s = (1000, 1000)
    a = stable_element_id("BUY", (100, 100, 80, 30), "button", s)
    b = stable_element_id("SELL", (100, 100, 80, 30), "button", s)
    assert a != b

def _el(id_: str, text: str, bbox: tuple[int, int, int, int], type_: str = "button") -> UIElement:
    return UIElement(
        id=id_, type=type_, text=text, bbox=bbox, confidence=0.9,
        sources=["vision", "ocr"], interactable=True,
    )


def test_diff_text_change() -> None:
    sde = StateDiffEngine()
    a = [_el("e1", "0.10", (100, 100, 80, 30), "input")]
    b = [_el("e1", "0.25", (100, 100, 80, 30), "input")]
    diff = sde.diff(a, b, "f1", "f2")
    assert "text_change" in diff.types()
    tc = next(c for c in diff.changes if c["type"] == "text_change")
    assert tc["before"] == "0.10" and tc["after"] == "0.25"


def test_diff_new_element() -> None:
    sde = StateDiffEngine()
    a: list[UIElement] = []
    b = [_el("e1", "Order Opened", (200, 300, 200, 60), "panel")]
    diff = sde.diff(a, b, "f1", "f2")
    assert "new_element" in diff.types()


def test_diff_removed_element() -> None:
    sde = StateDiffEngine()
    a = [_el("e1", "X", (10, 10, 20, 20), "button")]
    b: list[UIElement] = []
    diff = sde.diff(a, b, "f1", "f2")
    assert "removed_element" in diff.types()


def test_diff_match_by_iou_when_id_drifts() -> None:
    """Same physical element, different text → still matched by IoU."""
    sde = StateDiffEngine()
    # Different IDs because text differs, but bbox identical → IoU=1.0 → match.
    a = [_el("id1", "BUY", (100, 100, 80, 30))]
    b = [_el("id2", "BUY!", (100, 100, 80, 30))]
    diff = sde.diff(a, b, "f1", "f2")
    types = diff.types()
    # Should NOT show as removed+new — should be a single text_change match.
    assert "text_change" in types
    assert "new_element" not in types
    assert "removed_element" not in types


def test_diff_match_by_fuzzy_text_when_bbox_drifts() -> None:
    """bbox far apart, but text similar → fuzzy matching kicks in."""
    sde = StateDiffEngine()
    a = [_el("id1", "Confirm Order", (100, 100, 200, 40))]
    b = [_el("id2", "Confirm Order", (500, 600, 200, 40))]  # IDs differ, IoU=0
    diff = sde.diff(a, b, "f1", "f2")
    types = diff.types()
    # Matched by fuzzy text → moved_element, not new+removed
    assert "moved_element" in types
    assert "new_element" not in types

def test_normalizer_produces_three_artifacts(synth_run: Path) -> None:
    from dare.normalizer.normalizer import Normalizer

    norm = Normalizer(run_dir=synth_run)
    summary = norm.run()
    proc = synth_run / "processed"
    assert (proc / "ui_state.json").is_file()
    assert (proc / "state_diff.json").is_file()
    assert (proc / "action_graph.json").is_file()
    assert summary["actions"] == 2  # one click + one type
    assert summary["frames"] >= 1


def test_normalizer_action_graph_links_targets(synth_run: Path) -> None:
    from dare.normalizer.normalizer import Normalizer

    Normalizer(run_dir=synth_run).run()
    graph = json.loads(
        (synth_run / "processed" / "action_graph.json").read_text("utf-8")
    )
    assert len(graph["actions"]) == 2
    click = graph["actions"][0]
    assert click["type"] == "click"
    # raw_xy preserved
    assert click["raw_xy"] == [150, 120]
    # relative position computed
    assert click["target"]["relative_position"] is not None
    typ = graph["actions"][1]
    assert typ["type"] == "type"
    assert typ["value"] == "0.25"
    # Type action declares text_change as expected effect
    assert typ.get("expected_diff") == "text_change"
